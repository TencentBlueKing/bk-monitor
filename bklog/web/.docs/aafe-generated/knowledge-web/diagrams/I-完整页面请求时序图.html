flowchart TD
  START([浏览器访问 /retrieve/:indexId]) --> HTML[index.html 静态壳]
  HTML --> BOOT[bootstrap-loading 首屏占位]
  HTML --> MAIN[main.js + preload]
  MAIN --> PRELOAD_API[并行: space / userInfo / globals]
  PRELOAD_API --> MOUNT[挂载 #app 并淡出 bootstrap-loading]
  MOUNT --> ROUTER[Vue Router]
  ROUTER --> HUB[RetrieveHub]
  HUB --> INIT[默认加载 retrieve-v3<br/>use-app-init]

  subgraph INITIALIZE[页面初始化]
    INIT --> PARAMS[写入 URL、空间、索引模式]
    PARAMS --> INDEX_LIST[dispatch getIndexSetList]
    INDEX_LIST --> INDEX_API[GET /search/index_set/]
    INDEX_API --> FIELD_ACTION[dispatch requestIndexSetFieldInfo]
    FIELD_ACTION --> FIELD_API[普通 / 联合 / 场景字段请求]
    FIELD_API --> FIELD_CACHE[normalize fields<br/>field cache + IndexedDB]
  end

  FIELD_CACHE --> QUERY_ACTION[dispatch requestIndexSetQuery]

  subgraph SEARCH[流式检索执行]
    QUERY_ACTION --> WORKER_SERVICE[Worker Service<br/>searchStream(queryKey, startSeq, writeMode)]
    WORKER_SERVICE --> WORKER[WebWorker<br/>postMessage search-stream]
    WORKER --> SEARCH_API[fetch POST /search/*?stream=true]
    SEARCH_API --> STREAM[NDJSON: meta / row / done<br/>兼容 JSON envelope]
    STREAM --> ROW_STORAGE[批量写入原始行与渲染元数据]
    ROW_STORAGE --> IDB[(IndexedDB retrieveRows)]
    STREAM --> PROGRESS[progress meta / first row / rowKeys]
    PROGRESS --> WORKER_SERVICE
  end

  WORKER_SERVICE --> STORE[Vuex<br/>loading / total / row_keys / timings]
  STORE --> RESULT[结果组件渲染首行和结果元数据]
  RESULT --> READ_ROWS[按 row_keys 读取行]
  READ_ROWS --> IDB
  IDB --> RENDER[原始行 + overlay 后的渲染行]
  RENDER --> RESULT

  RESULT --> LOAD_MORE[用户加载更多]
  LOAD_MORE --> PAGE_QUERY[分页 query(startSeq, append)]
  PAGE_QUERY --> WORKER_SERVICE
  WORKER_SERVICE --> DONE[完成 / 错误 / 取消]
  DONE --> STORE
