flowchart TD
  START([进入 /retrieve]) --> HUB[retrieve-hub]
  HUB --> VERSION{retrieve_version}
  VERSION -->|默认或非 v1| V3[retrieve-v3]
  VERSION -->|v1| LEGACY[兼容旧版 retrieve/container]
  V3 --> URL[use-app-init 解析 URL]
  URL --> TYPE{索引/检索类型}
  TYPE -->|index_id| SINGLE[单索引 ids]
  TYPE -->|unionList 或 tags| UNION[联合索引 ids]
  TYPE -->|scene| SCENE[场景检索参数]
  SINGLE --> LIST[retrieve/getIndexSetList]
  UNION --> LIST
  SCENE --> LIST
  LIST --> VALIDATE{索引集可用?}
  VALIDATE -->|否| UNAUTH[跳转无权限页]
  VALIDATE -->|是| CONFIG{场景检索?}
  CONFIG -->|是| SCENECONFIG[requestSceneConfigs<br/>URL 回填筛选值]
  CONFIG -->|否| FIELD
  SCENECONFIG --> EMPTY{场景筛选为空?}
  EMPTY -->|是| STOP[跳过查询并展示空状态]
  EMPTY -->|否| FIELD[requestIndexSetFieldInfo]
  FIELD --> BRANCH{字段接口分支}
  BRANCH -->|普通| NORMALFIELD[GET /search/index_set/:id/fields/]
  BRANCH -->|联合| UNIONFIELD[unionSearch/unionMapping]
  BRANCH -->|场景| SCENEFIELD[POST /search/scene/fields/]
  NORMALFIELD --> CACHE[规范化并写入字段缓存]
  UNIONFIELD --> CACHE
  SCENEFIELD --> CACHE
  CACHE --> TAB{origin tab 且有字段?}
  TAB -->|是| QUERY[requestIndexSetQuery]
  TAB -->|否| RESULTTAB[由结果 tab 自行加载]
  QUERY --> END([渲染结果])
  RESULTTAB --> END
